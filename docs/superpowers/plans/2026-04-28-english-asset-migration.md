# English Asset Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the read-only inventory, manifest validation, and subagent batch-preparation pipeline for the full Korean-to-English asset migration.

**Architecture:** This first plan intentionally does not rename, copy, delete, or patch game assets. It creates a small Python standard-library tool under `tools/asset_migration/` that scans the workspace, writes CSV/Markdown reports under `docs/asset-migration/`, validates translation manifests, and prepares safe subagent input batches. Later plans will use the generated manifests to apply overlays and update references.

**Tech Stack:** Python 3 standard library (`argparse`, `csv`, `dataclasses`, `pathlib`, `re`, `unittest`), PowerShell command execution, existing Korangar/rAthena workspace.

---

## Scope Boundary

This plan implements Iteration 1 and Iteration 2 from `docs/superpowers/specs/2026-04-28-english-asset-migration-design.md`.

- Read-only inventory scanner.
- Read-only text/reference scanner.
- Manifest data model and validator.
- Batch CSV generator for subagents.

- Renaming files.
- Creating `archive_en/`.
- Editing `.spr`, `.act`, `.rsm`, `.rsw`, `.lub`, `.grf`, or rAthena scripts.
- Updating `client/game_archives.ron`.
- Removing Korean aliases.

```python
from __future__ import annotations

import re
from pathlib import PureWindowsPath

HANGUL_RE = re.compile(r"[\uac00-\ud7a3]")
MOJIBAKE_RE = re.compile(r"(�|Ã|Â|ë|ì|í|ð|þ)")
INVALID_WINDOWS_PATH_CHARS_RE = re.compile(r'[<>:"|?*]')


def has_hangul(value: str) -> bool:
    return HANGUL_RE.search(value) is not None


def has_mojibake_marker(value: str) -> bool:
    return MOJIBAKE_RE.search(value) is not None


def normalize_asset_path(value: str) -> str:
    return str(PureWindowsPath(value.replace("/", "\\"))).lower()


def is_safe_ascii_target_path(value: str) -> bool:
    if value.startswith("..") or "\\..\\" in value or "/../" in value:
        return False
    if INVALID_WINDOWS_PATH_CHARS_RE.search(value):
        return False
    try:
        value.encode("ascii")
    except UnicodeEncodeError:
        return False
    return True


def classify_asset_kind(asset_path: str) -> str:
    normalized = normalize_asset_path(asset_path)
    extension = PureWindowsPath(normalized).suffix.lower()

    if normalized.endswith("sclientinfo.xml") or normalized.endswith(".ron"):
        return "client-ui-and-config"

    if "\\lua files\\" in normalized or "\\luafiles514\\" in normalized or extension in {".lua", ".lub"}:
        return "lua-lub-visible-text"

    if "\\texture\\" in normalized or "\\유저인터페이스\\" in normalized or extension in {".bmp", ".jpg", ".jpeg", ".png", ".tga"}:
        return "texture-visible-text"

    if "\\sprite\\" in normalized or extension in {".spr", ".act", ".pal", ".imf"}:
        return "sprite-job-monster-npc-paths"

    if extension in {".rsw", ".rsm", ".rsm2", ".gnd", ".gat"}:
        return "map-model-reference-paths"

    return "uncategorized"
```

```python
from __future__ import annotations

import csv
from pathlib import Path
from typing import Iterable, Mapping


def ensure_parent(path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)


def write_csv(path: Path, header: list[str], rows: Iterable[Mapping[str, object]]) -> None:
    ensure_parent(path)
    with path.open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=header, extrasaction="ignore")
        writer.writeheader()
        for row in rows:
            writer.writerow(row)


def read_csv(path: Path) -> list[dict[str, str]]:
    with path.open("r", encoding="utf-8", newline="") as handle:
        return list(csv.DictReader(handle))
```

```python
from __future__ import annotations

from collections import Counter
from pathlib import Path

from .io_utils import write_csv
from .models import FILE_INVENTORY_HEADER, FileInventoryRow
from .path_rules import classify_asset_kind, has_hangul, has_mojibake_marker, normalize_asset_path


ARCHIVE_DATA_RELATIVE = Path("korangar") / "korangar" / "archive" / "data"


def build_inventory(root: Path) -> list[FileInventoryRow]:
    archive_data = root / ARCHIVE_DATA_RELATIVE
    if not archive_data.exists():
        return []

    rows: list[FileInventoryRow] = []
    for file in archive_data.rglob("*"):
        if not file.is_file():
            continue

        asset_path = normalize_asset_path(str(Path("data") / file.relative_to(archive_data)))
        rows.append(
            FileInventoryRow(
                asset_path=asset_path,
                absolute_path=str(file),
                extension=file.suffix.lower(),
                size_bytes=file.stat().st_size,
                kind=classify_asset_kind(asset_path),
                has_hangul=has_hangul(asset_path),
                has_mojibake=has_mojibake_marker(asset_path),
            )
        )

    rows.sort(key=lambda row: row.asset_path)
    return rows


def summarize_inventory(rows: list[FileInventoryRow]) -> str:
    total = len(rows)
    korean = [row for row in rows if row.has_hangul]
    mojibake = [row for row in rows if row.has_mojibake]
    by_kind = Counter(row.kind for row in korean)
    by_extension = Counter(row.extension or "<none>" for row in korean)

    lines = [
        "# Asset Migration Impact Summary",
        "",
        f"Total files: {total}",
        f"Korean path files: {len(korean)}",
        f"Mojibake path files: {len(mojibake)}",
        "",
        "## Korean Paths By Kind",
        "",
    ]

    for kind, count in sorted(by_kind.items()):
        lines.append(f"- {kind}: {count}")

    lines.extend(["", "## Korean Paths By Extension", ""])
    for extension, count in sorted(by_extension.items()):
        lines.append(f"- {extension}: {count}")

    lines.append("")
    return "\n".join(lines)


def write_inventory_outputs(root: Path, output_dir: Path) -> None:
    rows = build_inventory(root)
    write_csv(output_dir / "files.csv", FILE_INVENTORY_HEADER, [row.to_csv_row() for row in rows])
    write_csv(
        output_dir / "korean-paths.csv",
        FILE_INVENTORY_HEADER,
        [row.to_csv_row() for row in rows if row.has_hangul],
    )
    (output_dir / "impact-summary.md").parent.mkdir(parents=True, exist_ok=True)
    (output_dir / "impact-summary.md").write_text(summarize_inventory(rows), encoding="utf-8")
```

```python
from __future__ import annotations

from pathlib import Path

from .io_utils import write_csv
from .models import REFERENCE_HEADER, ReferenceRow
from .path_rules import has_hangul, has_mojibake_marker

TEXT_EXTENSIONS = {
    ".bat",
    ".conf",
    ".cpp",
    ".h",
    ".hpp",
    ".lua",
    ".md",
    ".ron",
    ".rs",
    ".txt",
    ".xml",
    ".yaml",
    ".yml",
}

SCAN_RELATIVE_ROOTS = [
    Path("korangar") / "korangar" / "archive" / "data",
    Path("korangar") / "korangar" / "client",
    Path("korangar") / "korangar" / "src",
    Path("rathena-master") / "npc",
    Path("rathena-master") / "db",
    Path("rathena-master") / "conf",
    Path("docker") / "import",
]


def iter_text_files(root: Path) -> list[Path]:
    files: list[Path] = []
    for relative in SCAN_RELATIVE_ROOTS:
        scan_root = root / relative
        if not scan_root.exists():
            continue
        for file in scan_root.rglob("*"):
            if file.is_file() and file.suffix.lower() in TEXT_EXTENSIONS:
                files.append(file)
    return sorted(files)


def read_text_lossy(path: Path) -> str:
    data = path.read_bytes()
    for encoding in ("utf-8", "euc-kr", "cp949", "latin-1"):
        try:
            return data.decode(encoding)
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace")


def scan_references(root: Path) -> list[ReferenceRow]:
    rows: list[ReferenceRow] = []
    for file in iter_text_files(root):
        text = read_text_lossy(file)
        relative_file = str(file.relative_to(root))
        for line_number, line in enumerate(text.splitlines(), start=1):
            if has_hangul(line):
                rows.append(
                    ReferenceRow(
                        source_file=relative_file,
                        line=line_number,
                        kind="hangul-text",
                        matched_text=line.strip()[:500],
                        notes="line contains Hangul",
                    )
                )
            elif has_mojibake_marker(line):
                rows.append(
                    ReferenceRow(
                        source_file=relative_file,
                        line=line_number,
                        kind="mojibake-text",
                        matched_text=line.strip()[:500],
                        notes="line contains common mojibake markers",
                    )
                )
    return rows


def write_reference_outputs(root: Path, output_path: Path) -> None:
    rows = scan_references(root)
    write_csv(output_path, REFERENCE_HEADER, [row.to_csv_row() for row in rows])
```

```python
from __future__ import annotations

from pathlib import Path

from .io_utils import read_csv
from .models import MANIFEST_HEADER
from .path_rules import is_safe_ascii_target_path

VALID_CONFIDENCE = {"low", "medium", "high"}


def parse_bool(value: str) -> bool:
    return value.strip().lower() in {"1", "yes", "y", "true"}


def validate_manifest(path: Path) -> list[str]:
    errors: list[str] = []
    rows = read_csv(path)

    missing_columns = [column for column in MANIFEST_HEADER if column not in (rows[0].keys() if rows else MANIFEST_HEADER)]
    for column in missing_columns:
        errors.append(f"missing required column: {column}")

    seen_targets: dict[str, int] = {}

    for index, row in enumerate(rows, start=2):
        source_path = row.get("source_path", "").strip()
        target_path = row.get("target_path", "").strip()
        confidence = row.get("confidence", "").strip().lower()
        needs_human_review = parse_bool(row.get("needs_human_review", ""))

        if not source_path:
            errors.append(f"line {index}: source_path is required")

        if not target_path:
            errors.append(f"line {index}: target_path is required")
        elif not target_path.lower().startswith("archive_en\\"):
            errors.append(f"line {index}: target_path must start with archive_en\\")
        elif not is_safe_ascii_target_path(target_path):
            errors.append(f"line {index}: target_path is not safe ASCII: {target_path}")

        if confidence not in VALID_CONFIDENCE:
            errors.append(f"line {index}: confidence must be low, medium, or high")

        if confidence == "low" and not needs_human_review:
            errors.append(f"line {index}: low confidence rows must request human review")

        target_key = target_path.lower()
        if target_key in seen_targets:
            errors.append(f"line {index}: duplicate target_path also used on line {seen_targets[target_key]}: {target_path}")
        else:
            seen_targets[target_key] = index

        if source_path and target_path:
            source_ext = Path(source_path).suffix.lower()
            target_ext = Path(target_path).suffix.lower()
            if source_ext != target_ext:
                errors.append(f"line {index}: target extension {target_ext} does not match source extension {source_ext}")

    return errors
```

```python
from __future__ import annotations

from pathlib import Path

from .inventory import build_inventory
from .io_utils import write_csv
from .models import MANIFEST_HEADER, FileInventoryRow

GLOSSARY_HEADER = ["source_term", "english_term", "category", "notes"]

BATCH_IDS = [
    "client-ui-and-config",
    "lua-lub-visible-text",
    "npc-dialog-and-cutin",
    "texture-visible-text",
    "sprite-job-monster-npc-paths",
    "map-model-reference-paths",
]


def build_batch_rows(batch_id: str, inventory_rows: list[FileInventoryRow]) -> list[dict[str, object]]:
    rows: list[dict[str, object]] = []
    for row in inventory_rows:
        if row.kind != batch_id:
            continue
        if not row.has_hangul and not row.has_mojibake:
            continue
        rows.append({
            "batch_id": batch_id,
            "kind": row.kind,
            "source_path": row.asset_path,
            "target_path": "",
            "source_text": "",
            "target_text": "",
            "confidence": "",
            "needs_human_review": "true",
            "notes": f"extension={row.extension};size_bytes={row.size_bytes}",
        })
    return rows


def build_glossary_rows() -> list[dict[str, str]]:
    return [
        {"source_term": "남", "english_term": "male", "category": "gender", "notes": "Often appears in sprite/job asset paths."},
        {"source_term": "여", "english_term": "female", "category": "gender", "notes": "Often appears in sprite/job asset paths."},
        {"source_term": "프론테라", "english_term": "prontera", "category": "city", "notes": "Use official Ragnarok romanization."},
        {"source_term": "페이욘", "english_term": "payon", "category": "city", "notes": "Use official Ragnarok romanization."},
        {"source_term": "모로코", "english_term": "morocc", "category": "city", "notes": "Use Ragnarok map spelling when it is a path."},
        {"source_term": "유저인터페이스", "english_term": "userinterface", "category": "folder", "notes": "Keep compact path spelling to match existing English folder."},
    ]


def write_batch_outputs(root: Path, output_dir: Path) -> None:
    inventory_rows = build_inventory(root)
    output_dir.mkdir(parents=True, exist_ok=True)

    write_csv(output_dir / "glossary.csv", GLOSSARY_HEADER, build_glossary_rows())

    for batch_id in BATCH_IDS:
        rows = build_batch_rows(batch_id, inventory_rows)
        write_csv(output_dir / f"{batch_id}.csv", MANIFEST_HEADER, rows)
```
